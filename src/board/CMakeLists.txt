# @req REQ-0049
include("${MCUBSP_ROOT}/cmake/Firmware.cmake")
if(NOT MCUBSP_BOARD MATCHES "^[a-z0-9_-]+/[a-z0-9_-]+$")
    message(FATAL_ERROR "Select MCUBSP_BOARD=vendor/name")
endif()
set(board_directory "${CMAKE_CURRENT_SOURCE_DIR}/${MCUBSP_BOARD}")
if(NOT EXISTS "${board_directory}/board.cmake")
    message(FATAL_ERROR "Unknown board: ${MCUBSP_BOARD}")
endif()
include("${board_directory}/board.cmake")
# Keep the selected-board interface available to existing board-local targets.
add_library(mcubsp_board INTERFACE)
add_library(mcubsp::board ALIAS mcubsp_board)
target_link_libraries(mcubsp_board INTERFACE mcubsp::io)
target_compile_options(mcubsp_board INTERFACE ${MCUBSP_ARCH_COMPILE_OPTIONS})
target_sources(mcubsp_board INTERFACE "${MCUBSP_ROOT}/${MCUBSP_BOARD_PROFILE_HEADER}")
set(MCUBSP_BOARD_REVISION "${MCUBSP_BOARD_DEFAULT_REVISION}" CACHE STRING "Exact board binding revision")
set(MCUBSP_STARTUP supplied CACHE STRING "Startup owner: supplied or external")
# Migrate cached repository example paths; external applications keep their source.
if(MCUBSP_APPLICATION AND NOT EXISTS "${MCUBSP_APPLICATION}")
    foreach(support_tree IN ITEMS board bsp)
        set(old_example_prefix "${MCUBSP_ROOT}/${support_tree}/${MCUBSP_BOARD}/example/")
        string(FIND "${MCUBSP_APPLICATION}" "${old_example_prefix}" old_example_position)
        if(old_example_position EQUAL 0)
            get_filename_component(example_name "${MCUBSP_APPLICATION}" NAME_WE)
            set(migrated_application "${MCUBSP_ROOT}/example/${example_name}/${MCUBSP_BOARD}.cpp")
            if(EXISTS "${migrated_application}")
                set(MCUBSP_APPLICATION "${migrated_application}" CACHE FILEPATH "Application source" FORCE)
            endif()
        endif()
    endforeach()
endif()
set(MCUBSP_APPLICATION "${MCUBSP_ROOT}/example/blink/${MCUBSP_BOARD}.cpp" CACHE FILEPATH "Application source")
mcubsp_add_firmware(firmware BOARD "${MCUBSP_BOARD}" REVISION "${MCUBSP_BOARD_REVISION}"
    STARTUP "${MCUBSP_STARTUP}" SOURCES "${MCUBSP_APPLICATION}")
if(EXISTS "${board_directory}/CMakeLists.txt")
    # Physical-test dependencies remain opt-in.
    add_subdirectory("${MCUBSP_BOARD}")
elseif(MCUBSP_BUILD_BOARD_TESTS)
    message(FATAL_ERROR "Use the board-local physical-test procedure for ${MCUBSP_BOARD}; the firmware target builds the application only")
endif()
