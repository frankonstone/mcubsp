cmake_minimum_required(VERSION 3.25)
project(my_system LANGUAGES CXX ASM)
set(MCUBSP_SOURCE_DIR "" CACHE PATH "mcubsp checkout; leave empty for an installed package")
if(MCUBSP_SOURCE_DIR)
    add_subdirectory("${MCUBSP_SOURCE_DIR}" mcubsp EXCLUDE_FROM_ALL)
else()
    find_package(mcubsp 0.1 CONFIG REQUIRED)
endif()
set(system_source "${CMAKE_CURRENT_SOURCE_DIR}/${MCUBSP_BOARD}.cpp")
if(NOT EXISTS "${system_source}")
    message(FATAL_ERROR "This system example has no source for ${MCUBSP_BOARD}")
endif()
mcubsp_add_firmware(system_demo STARTUP supplied SOURCES "${system_source}")
